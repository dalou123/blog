    前端大文件上传

    dalou 2024/5/17 javascript

    # 前端大文件上传

    在前端实现大文件上传时，直接使用传统的文件上传方式可能会导致内存占用过高、上传失败或用户体验差。为了解决这些问题，可以采用分片上传、断点续传、并行上传等技术。

    # 分片上传

    将大文件分割成多个小块（chunks），分别上传到服务器，最后在服务器端合并。

    # 实现步骤：

    1. 前端分片：

    • 使用 File 对象的 slice 方法将文件分割成多个小块。
    • 为每个分片生成唯一标识（如 MD5 哈希）。
    • 使用 FormData 或 fetch 上传分片。 2. 后端合并：
    • 服务器接收分片并存储。
    • 所有分片上传完成后，合并分片为完整文件。

    # 示例代码：

    // 分片大小（例如 1MB）
const CHUNK_SIZE = 1024 * 1024;

// 上传文件
async function uploadFile(file) {
  const fileSize = file.size;
  let start = 0;
  let chunkIndex = 0;

  while (start < fileSize) {
    const chunk = file.slice(start, start + CHUNK_SIZE);
    const formData = new FormData();
    formData.append('file', chunk);
    formData.append('chunkIndex', chunkIndex);
    formData.append('totalChunks', Math.ceil(fileSize / CHUNK_SIZE));
    formData.append('fileName', file.name);

    await uploadChunk(formData);

    start += CHUNK_SIZE;
    chunkIndex++;
  }

  console.log('文件上传完成');
}

// 上传分片
async function uploadChunk(formData) {
  return fetch('/upload-chunk', {
    method: 'POST',
    body: formData,
  }).then(response => response.json());
}

    # 断点续传

    在上传过程中，如果网络中断或用户暂停，可以从上次中断的位置继续上传。

    # 实现步骤：

    1. 记录上传进度：

    • 使用 localStorage 或 IndexedDB 存储已上传的分片信息。 2. 恢复上传：
    • 上传前检查已上传的分片，跳过已上传的部分。

    # 示例代码：

    // 获取已上传的分片信息
function getUploadedChunks(fileName) {
  const uploadedChunks = localStorage.getItem(fileName) || '[]';
  return JSON.parse(uploadedChunks);
}

// 更新已上传的分片信息
function updateUploadedChunks(fileName, chunkIndex) {
  const uploadedChunks = getUploadedChunks(fileName);
  uploadedChunks.push(chunkIndex);
  localStorage.setItem(fileName, JSON.stringify(uploadedChunks));
}

// 上传文件（支持断点续传）
async function uploadFile(file) {
  const fileSize = file.size;
  const totalChunks = Math.ceil(fileSize / CHUNK_SIZE);
  const uploadedChunks = getUploadedChunks(file.name);

  for (let chunkIndex = 0; chunkIndex < totalChunks; chunkIndex++) {
    if (uploadedChunks.includes(chunkIndex)) {
      console.log(`分片 ${chunkIndex} 已上传，跳过`);
      continue;
    }

    const start = chunkIndex * CHUNK_SIZE;
    const chunk = file.slice(start, start + CHUNK_SIZE);
    const formData = new FormData();
    formData.append('file', chunk);
    formData.append('chunkIndex', chunkIndex);
    formData.append('totalChunks', totalChunks);
    formData.append('fileName', file.name);

    await uploadChunk(formData);
    updateUploadedChunks(file.name, chunkIndex);
  }

  console.log('文件上传完成');
  localStorage.removeItem(file.name); // 清理记录
}

    # 并行上传

    同时上传多个分片，提升上传速度。

    # 实现步骤：

    • 使用 Promise.all 或 Worker 实现并行上传。
    • 控制并发数，避免过多请求导致浏览器崩溃。

    示例代码：

    // 并行上传分片
async function uploadFile(file) {
  const fileSize = file.size;
  const totalChunks = Math.ceil(fileSize / CHUNK_SIZE);
  const uploadPromises = [];

  for (let chunkIndex = 0; chunkIndex < totalChunks; chunkIndex++) {
    const start = chunkIndex * CHUNK_SIZE;
    const chunk = file.slice(start, start + CHUNK_SIZE);
    const formData = new FormData();
    formData.append('file', chunk);
    formData.append('chunkIndex', chunkIndex);
    formData.append('totalChunks', totalChunks);
    formData.append('fileName', file.name);

    uploadPromises.push(uploadChunk(formData));
  }

  await Promise.all(uploadPromises);
  console.log('文件上传完成');
}

    # 进度显示

    实时显示上传进度，提升用户体验。

    # 实现步骤：

    • 监听 XMLHttpRequest 或 fetch 的进度事件。
    • 计算并更新进度条。

    示例代码：

    // 上传分片并显示进度
function uploadChunk(formData) {
  return new Promise((resolve, reject) => {
    const xhr = new XMLHttpRequest();
    xhr.open('POST', '/upload-chunk', true);

    xhr.upload.onprogress = (event) => {
      if (event.lengthComputable) {
        const percent = (event.loaded / event.total) * 100;
        console.log(`上传进度: ${percent.toFixed(2)}%`);
      }
    };

    xhr.onload = () => resolve(xhr.responseText);
    xhr.onerror = () => reject(xhr.statusText);

    xhr.send(formData);
  });
}

    # 后端实现

    后端需要支持分片上传和合并文件

    示例（Node.js + Express）：

    const express = require('express');
const fs = require('fs');
const path = require('path');
const app = express();

app.use(express.json());

// 上传分片
app.post('/upload-chunk', (req, res) => {
  const { fileName, chunkIndex, totalChunks } = req.body;
  const chunk = req.files.file;
  const chunkDir = path.join(__dirname, 'chunks', fileName);

  if (!fs.existsSync(chunkDir)) {
    fs.mkdirSync(chunkDir, { recursive: true });
  }

  fs.writeFileSync(path.join(chunkDir, chunkIndex), chunk.data);
  res.json({ success: true });
});

// 合并分片
app.post('/merge-chunks', (req, res) => {
  const { fileName, totalChunks } = req.body;
  const chunkDir = path.join(__dirname, 'chunks', fileName);
  const outputFile = path.join(__dirname, 'uploads', fileName);

  const writeStream = fs.createWriteStream(outputFile);

  for (let i = 0; i < totalChunks; i++) {
    const chunkPath = path.join(chunkDir, i.toString());
    const chunkData = fs.readFileSync(chunkPath);
    writeStream.write(chunkData);
    fs.unlinkSync(chunkPath); // 删除分片
  }

  writeStream.end();
  res.json({ success: true });
});

app.listen(3000, () => {
  console.log('Server is running on port 3000');
});
