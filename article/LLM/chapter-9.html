    # 九、长期记忆

    大模型每次调用都是无状态的——它不记得上次说了什么，不知道你是谁，不了解你的背景。长期记忆解决这个问题，让 AI 应用在多次对话之间保持连贯，提供真正个性化的体验。

    # 9.1 记忆的分类

    实现 AI 记忆系统之前，先明确要记什么：

    类型 存什么 例子
    会话记忆（短期） 当前对话的历史 "你刚才说你在学 Vue3"
    用户画像 用户的属性和偏好 技术水平、偏好语言、当前目标
    情节记忆 发生过的事件 "上周用户提到在做微前端迁移"
    语义记忆 积累的知识 "用户掌握了 Pinia 的基本用法"
    偏好记忆 用户的喜好 "不喜欢 TypeScript，偏好简短回答"

    # 9.2 会话记忆（短期）

    # 三种策略

    全量历史

    把所有历史消息都传给模型，记忆最完整，但 token 随对话轮次线性增长：

    滑动窗口

    只保留最近 N token 范围内的历史，超出从最老的成对删除：

    class SlidingWindowChat {
  constructor({ maxTokens = 3000 } = {}) {
    this.history = []
    this.maxTokens = maxTokens
  }

  estTokens() {
    return this.history.reduce(
      (s, m) => s + Math.ceil((m.content?.length ?? 0) * 0.6), 0
    )
  }

  trim() {
    // 成对删除（user + assistant），保持消息对齐
    while (this.history.length > 2 && this.estTokens() > this.maxTokens) {
      this.history.splice(0, 2)
    }
  }

  async chat(input) {
    this.history.push(new HumanMessage(input))
    this.trim()
    const res = await model.invoke([new SystemMessage(this.systemPrompt), ...this.history])
    this.history.push(new AIMessage(res.content))
    return res.content
  }
}
// 适合：长对话、对最近上下文要求高的场景

    摘要压缩

    历史超过阈值时，用模型把老消息压缩成摘要，注入 system，保留最近几轮原文：

    class SummaryChat {
  constructor({ summaryThreshold = 8 } = {}) {
    this.history = []
    this.summary = ''
    this.summaryThreshold = summaryThreshold
  }

  async compress() {
    if (this.history.length < this.summaryThreshold) return

    const oldMessages = this.history.slice(0, -4)
    const histText = oldMessages
      .map(m => `${m._getType() === 'human' ? 'U' : 'A'}: ${m.content}`)
      .join('\n')

    const res = await model.invoke([
      new SystemMessage('将对话历史压缩为简洁摘要，保留关键信息，不超过100字。'),
      new HumanMessage(histText),
    ])

    this.summary = res.content
    this.history = this.history.slice(-4)  // 只保留最近 2 轮原文
  }

  async chat(input) {
    if (this.history.length >= this.summaryThreshold) await this.compress()

    this.history.push(new HumanMessage(input))

    const sys = this.summary
      ? `${this.systemPrompt}\n\n历史背景：${this.summary}`
      : this.systemPrompt

    const res = await model.invoke([new SystemMessage(sys), ...this.history])
    this.history.push(new AIMessage(res.content))
    return res.content
  }
}
// 适合：超长对话、需要保留语义但不要求精确引用的场景

    # RunnableWithMessageHistory

    LangChain.js 内置的会话管理，自动注入历史，多用户完全隔离：

    import { RunnableWithMessageHistory } from '@langchain/core/runnables'
import { InMemoryChatMessageHistory } from '@langchain/core/chat_history'
import { MessagesPlaceholder } from '@langchain/core/prompts'

const histories = {}  // 生产换 Redis 或数据库

const chain = new RunnableWithMessageHistory({
  runnable: ChatPromptTemplate.fromMessages([
    ['system', '你是前端学习助手。'],
    new MessagesPlaceholder('history'),  // 历史自动注入
    ['human', '{input}'],
  ]).pipe(model).pipe(new StringOutputParser()),
  getMessageHistory: (sessionId) => {
    if (!histories[sessionId]) histories[sessionId] = new InMemoryChatMessageHistory()
    return histories[sessionId]

  historyMessagesKey: 'history',
})

const aliceConfig = { configurable: { sessionId: 'alice' } }
await chain.invoke({ input: '我在学 Vue3' }, aliceConfig)
const r = await chain.invoke({ input: '我之前说在学什么？' }, aliceConfig)

    # 从对话中提取

    每轮对话后，用结构化输出提取新信息，增量更新画像：

    const ExtractSchema = z.object({
  hasNewInfo: z.boolean(),
  updates: z.object({
    name:           z.string().optional(),
    techLevel:      z.enum(['beginner', 'intermediate', 'senior']).optional(),
    primaryStack:   z.array(z.string()).optional(),
    currentGoal:    z.string().optional(),
    newKnownTopics: z.array(z.string()).optional(),
    prefersCode:    z.boolean().optional(),
  }).optional(),
})

async function extractFromConversation(userMsg, aiMsg, currentProfile) {
  const result = await model.withStructuredOutput(ExtractSchema).invoke([
    new SystemMessage(`从对话中提取用户信息，只提取有明确依据的，不要猜测。
当前画像：${JSON.stringify(currentProfile)}`),
    new HumanMessage(`用户：${userMsg}\nAI：${aiMsg.slice(0, 200)}`),
  ])

  if (!result.hasNewInfo || !result.updates) return currentProfile

  const updated = { ...currentProfile }
  const u = result.updates

  if (u.name)        updated.name = u.name
  if (u.techLevel)   updated.techLevel = u.techLevel
  if (u.currentGoal) updated.currentGoal = u.currentGoal

  // 数组字段去重追加
  if (u.primaryStack?.length) {
    updated.primaryStack = [...new Set([...currentProfile.primaryStack, ...u.primaryStack])]
  }
  if (u.newKnownTopics?.length) {
    updated.knownTopics = [...new Set([...currentProfile.knownTopics, ...u.newKnownTopics])]
  }

  return updated
}

    # 画像注入对话

    function profileToContext(profile) {
  const parts = []
  if (profile.name) parts.push(`用户：${profile.name}`)
  if (profile.techLevel) {
    const map = { beginner: '初级', intermediate: '中级', senior: '资深' }
    parts.push(`水平：${map[profile.techLevel]}`)
  }
  if (profile.primaryStack?.length) parts.push(`技术栈：${profile.primaryStack.join(', ')}`)
  if (profile.currentGoal) parts.push(`目标：${profile.currentGoal}`)
  if (profile.knownTopics?.length) parts.push(`已掌握：${profile.knownTopics.slice(0, 5).join(', ')}`)
  if (profile.prefersCode) parts.push('偏好：代码示例')

  return parts.length ? `\n\n用户背景：\n${parts.map(p => `- ${p}`).join('\n')}` : ''
}

async function chat(userId, userInput) {
  const profile = await loadProfile(userId)
  const ctx = profileToContext(profile)

  const res = await model.invoke([
    new SystemMessage(`你是前端导师，根据用户背景个性化回答。${ctx}`),
    ...history,
  ])

  // 异步更新画像，不阻塞回复
  extractFromConversation(userInput, res.content, profile)
    .then(updated => saveProfile(userId, updated))
    .catch(() => {})

  return res.content
}

    # 9.4 长期记忆（跨会话）

    用向量数据库存储重要信息，按语义相关性检索，实现跨会话记忆。

    # 存取记忆

    import { MemoryVectorStore } from 'langchain/vectorstores/memory'
import { Document } from '@langchain/core/documents'

const vectorStore = new MemoryVectorStore(embeddings)

// 存入
async function storeMemory(userId, content, memType) {
  await vectorStore.addDocuments([
    new Document({
      pageContent: content,
      metadata: { userId, memType, createdAt: new Date().toISOString() },
    }),
  ])
}

// 检索（按语义相关性）
async function recallMemory(userId, query, k = 3) {
  const docs = await vectorStore.similaritySearchWithScore(query, k)
  return docs
    .filter(([, score]) => score > 0.3)   // 过滤低相关性
    .map(([doc, score]) => ({
      content: doc.pageContent,
      type: doc.metadata.memType,
      score,
    }))
}

    # 完整的跨会话流程

    class MemoryAugmentedChat {
  constructor(userId) {
    this.userId = userId
    this.shortTermHistory = []
  }

  async chat(input) {
    // 检索相关长期记忆
    const memories = await recallMemory(this.userId, input)
    const memCtx = memories.length
      ? '\n相关记忆：\n' + memories.map(m => `- [${m.type}] ${m.content}`).join('\n')
      : ''

    this.shortTermHistory.push(new HumanMessage(input))

    const res = await model.invoke([
      new SystemMessage(`你是前端助手。${memCtx}`),
      ...this.shortTermHistory.slice(-6),  // 短期只保留最近 3 轮
    ])

    this.shortTermHistory.push(new AIMessage(res.content))

    // 判断是否存入长期记忆
    const { shouldStore, memoryType, summary } = await shouldRemember(input, res.content)
    if (shouldStore && summary) {
      await storeMemory(this.userId, summary, memoryType)
    }

    return res.content
  }

  newSession() {
    this.shortTermHistory = []  // 清空短期，长期记忆保留
  }
}

// 验证跨会话效果
const chat = new MemoryAugmentedChat('bob')
await chat.chat('我在做 Vue2 到 Vue3 的迁移，时间比较紧')

chat.newSession()  // 新会话，短期清空

const r = await chat.chat('迁移过程中最容易踩什么坑？')
// AI 回答会结合"时间紧、Vue2→Vue3 迁移"的背景

    # 9.5 分层记忆架构

    把记忆分三层，各层有不同的生命周期和淘汰策略：

    L1（工作记忆）  最近 6 条消息，精确原文，随窗口滚动淘汰
      ↓ 每 N 轮压缩
L2（会话摘要）  当前会话的语义压缩，80 字以内
      ↓ 重要信息提升
L3（长期记忆）  核心事实，向量存储，跨会话永久保留

    class LayeredMemory {
  constructor() {
    this.l1 = []      // 最近 6 条消息
    this.l2 = ''      // 会话摘要
    this.l3 = []      // 核心事实条目
    this.turnCount = 0
  }

  async updateL2() {
    const histText = this.l1
      .map(m => `${m._getType() === 'human' ? 'U' : 'A'}: ${m.content}`)
      .join('\n')

    const res = await model.invoke([
      new SystemMessage('把对话总结成摘要，保留用户相关信息，不超过80字。'),
      new HumanMessage(histText),
    ])
    this.l2 = res.content
  }

  async chat(input, systemPrompt) {
    this.turnCount++
    if (this.turnCount % 4 === 0) await this.updateL2()

    // 三层记忆组合成上下文
    const memParts = []
    if (this.l3.length) memParts.push(`长期记忆：${this.l3.join('; ')}`)
    if (this.l2)        memParts.push(`本次会话：${this.l2}`)

    const sys = memParts.length
      ? `${systemPrompt}\n\n${memParts.join('\n')}`
      : systemPrompt

    this.l1.push(new HumanMessage(input))
    if (this.l1.length > 6) this.l1.shift()

    const res = await model.invoke([new SystemMessage(sys), ...this.l1.slice(-4)])
    this.l1.push(new AIMessage(res.content))
    if (this.l1.length > 6) this.l1.shift()

    return res.content
  }

  promoteToL3(fact) {
    this.l3.push(fact)  // 手动/自动把重要事实提升到 L3
  }
}

    # 9.6 Token 感知的历史截取

    不同内容的 token 消耗差异很大，按字符数截断不够精确：

    function estimateTokens(text) {
  const cnChars = (text.match(/[\u4e00-\u9fff]/g) || []).length
  const otherChars = text.length - cnChars
  return Math.ceil(cnChars * 0.6 + otherChars * 0.25)
}

// 从最新消息往前，放不下就停
function fitInWindow(messages, maxTokens) {
  const result = []
  let total = 0

  for (let i = messages.length - 1; i >= 0; i--) {
    const t = estimateTokens(messages[i].content || '')
    if (total + t > maxTokens) break
    result.unshift(messages[i])
    total += t
  }

  return result
}

// 使用：加载历史时自动截到窗口限制
const trimmed = fitInWindow(fullHistory, 2000)
const res = await model.invoke([systemMessage, ...trimmed, newUserMessage])

    app.post('/api/chat/stream', async (req, res) => {

  // 1. 并行加载：画像 + 长期记忆

    recallMemory(userId, message),
  ])

  // 2. 构建个性化 system
  const systemParts = ['你是前端开发助手。']
  const ctx = profileToContext(profile)
  if (ctx) systemParts.push(ctx)
  if (memories.length) {
    systemParts.push('相关记忆：\n' + memories.map(m => `- ${m.content}`).join('\n'))
    // 通知前端：AI 引用了哪些历史记忆
    send('memories', { count: memories.length, items: memories.map(m => m.content.slice(0, 60)) })
  }

  // 3. 流式生成
  const history = getSessionHistory(sessionId)
  const trimmed = fitInWindow(history, 2000)
  let fullReply = ''

  for await (const chunk of model.stream([
    new SystemMessage(systemParts.join('\n\n')),
    ...trimmed,
    new HumanMessage(message),
  ])) {
    fullReply += chunk.content
    send('token', { token: chunk.content })
  }

  history.push(new HumanMessage(message))
  history.push(new AIMessage(fullReply))

  // 4. 异步更新记忆（不阻塞响应）
  updateMemoryAsync(userId, message, fullReply, profile).catch(() => {})

  send('done', {})
  res.end()
})

    # 9.9 工程注意事项

    记忆准确性问题：AI 提取的信息可能有误，让用户能查看和编辑画像，提供"删除这条记忆"的入口。

    记忆过多的问题：长期使用后记忆条目积累，检索质量下降。处理方式：按 importance 字段定期清理低分记忆，或限制每用户上限（如 500 条）。

    隐私问题：在界面展示存储的内容，隐私政策里说明，提供清空全部记忆的功能。

    # 9.10 本章小结

    • 会话记忆三种方案：全量（短对话）、滑动窗口（长对话）、摘要压缩（超长对话）；RunnableWithMessageHistory 是最省事的内置方案
    • 用户画像从对话中增量提取，数组字段去重追加，每次对话自动注入 system 提示
    • 长期记忆用向量存储，每轮结束后判断是否值得存，检索时过滤 score < 0.3 的低相关结果
    • 分层记忆：L1 工作记忆（精确短命）→ L2 会话摘要（压缩中期）→ L3 长期记忆（核心永久）
    • Token 感知截取：中文按 0.6 token/字，其他按 0.25 token/字符估算，从最新消息往前装
    • 前端展示记忆引用增加透明度，提供用户管理记忆的入口
    • 生产环境：向量库用 Chroma/Pinecone，画像用 PostgreSQL，不能用进程内存储
