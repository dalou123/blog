    # 八、Workflow 工作流

    第七章的 Agent 适合任务路径不确定、需要模型自主规划的场景。Workflow 适合另一类场景：执行步骤已知、顺序固定、需要稳定可控——文章生成、代码审查、数据处理管道都属于这类。

    # 8.1 Workflow vs Agent

    两者的核心区别是谁来决定执行路径：

    Workflow Agent
    执行路径 开发者预先定义 模型自主决定
    适合任务 步骤明确、可重复 任务复杂、路径不确定
    可预测性 高 低
    调试难度 低 高
    典型场景 内容生成流程、数据处理管道 开放式问答、多工具协作

    实际项目里两者常结合用：Workflow 定义整体骨架和确定步骤，Agent 处理其中需要灵活判断的环节。

    # 8.2 线性工作流

    最简单的形式：固定顺序，每步输入来自上一步的输出。

    import { StateGraph, END, START, Annotation } from '@langchain/langgraph'
import { ChatPromptTemplate } from '@langchain/core/prompts'
import { StringOutputParser } from '@langchain/core/output_parsers'

// 定义贯穿整个工作流的状态
const State = Annotation.Root({
  topic:     Annotation({ reducer: (_, n) => n, default: () => '' }),
  analysis:  Annotation({ reducer: (_, n) => n, default: () => '' }),
  outline:   Annotation({ reducer: (_, n) => n, default: () => '' }),
  draft:     Annotation({ reducer: (_, n) => n, default: () => '' }),
  finalPost: Annotation({ reducer: (_, n) => n, default: () => '' }),
})

// 节点1：分析选题
async function analyzeNode(state) {
  const chain = ChatPromptTemplate.fromMessages([
    ['system', '技术内容策略专家，分析受众需求和文章价值点，回答简洁。'],
    ['human', '主题：{topic}\n\n分析：读者痛点、核心价值、差异化角度'],
  ]).pipe(model).pipe(new StringOutputParser())

  const analysis = await chain.invoke({ topic: state.topic })
  return { analysis }  // 只返回变化的字段
}

// 节点2：生成大纲（依赖上一步的 analysis）
async function outlineNode(state) {
  const chain = ChatPromptTemplate.fromMessages([
    ['system', '技术写作专家，生成清晰的文章大纲。'],
    ['human', '主题：{topic}\n选题分析：{analysis}\n\n生成 4-5 章节大纲，每章含 2-3 个要点'],
  ]).pipe(model).pipe(new StringOutputParser())

  const outline = await chain.invoke({ topic: state.topic, analysis: state.analysis })
  return { outline }
}

// 节点3：撰写草稿（依赖 outline）
async function draftNode(state) { ... }

// 节点4：润色（依赖 draft）
async function polishNode(state) { ... }

// 构建线性工作流
const workflow = new StateGraph(State)
  .addNode('analyze', analyzeNode)
  .addNode('outline', outlineNode)
  .addNode('draft',   draftNode)
  .addNode('polish',  polishNode)
  .addEdge(START,     'analyze')  // 固定顺序
  .addEdge('analyze', 'outline')
  .addEdge('outline', 'draft')
  .addEdge('draft',   'polish')
  .addEdge('polish',   END)
  .compile()

const result = await workflow.invoke({
  topic: 'Vue3 Composition API 实战指南',
})
console.log(result.finalPost)

    # 8.3 条件工作流

    根据中间步骤的结果，动态选择不同的执行路径。

    代码输入
    ↓
[分类节点]：判断代码类型和复杂度
    ↓
条件路由 ─────┬── review_component（组件代码：性能+风格）
              ├── review_logic（工具/API：逻辑+安全）
              └── review_general（其他：通用审查）
                         ↓
              [generate_report]：汇聚三条路径的结果

    // 分类节点：用结构化输出获得明确的分类结果
const ClassifySchema = z.object({
  codeType: z.enum(['component', 'utility', 'api', 'config', 'test']),
  complexity: z.enum(['low', 'medium', 'high']),
})

async function classifyNode(state) {
  const result = await model.withStructuredOutput(ClassifySchema).invoke([
    { role: 'system', content: '分析代码类型和复杂度。' },
    { role: 'user', content: `分析：\n${state.code}` },
  ])
  return { codeType: result.codeType, complexity: result.complexity }
}

// 路由函数：返回下一个节点名称
function routeByType(state) {
  if (state.codeType === 'component') return 'review_component'
  if (['utility', 'api'].includes(state.codeType)) return 'review_logic'
  return 'review_general'
}

const reviewWorkflow = new StateGraph(State)
  .addNode('classify',         classifyNode)
  .addNode('review_component', reviewComponentNode)
  .addNode('review_logic',     reviewLogicNode)
  .addNode('review_general',   reviewGeneralNode)
  .addNode('generate_report',  generateReportNode)
  .addEdge(START, 'classify')
  // 条件边：classify 执行完后调用 routeByType 决定走哪条路
  .addConditionalEdges('classify', routeByType, {
    review_component: 'review_component',
    review_logic:     'review_logic',
    review_general:   'review_general',
  })
  // 三条路径都汇聚到 generate_report（扇入）
  .addEdge('review_component', 'generate_report')
  .addEdge('review_logic',     'generate_report')
  .addEdge('review_general',   'generate_report')
  .addEdge('generate_report',   END)
  .compile()

    # 8.4 并行工作流

const State = Annotation.Root({

const parallelWorkflow = new StateGraph(State)
  .addNode('perf',  perfNode)
  .addNode('a11y',  a11yNode)

    # 8.5 Human-in-the-Loop（人工干预）

    在关键节点暂停工作流，等待人工审核或修改，然后继续。

    import { MemorySaver } from '@langchain/langgraph'

const checkpointer = new MemorySaver()  // 生产环境换 PostgresSaver 或 RedisSaver

const workflow = new StateGraph(State)
  .addNode('generate_draft', generateDraftNode)
  .addNode('human_review',   humanReviewNode)   // 中断点
  .addNode('revise',         reviseNode)
  .addNode('publish',        publishNode)
  // ...边定义...
  .compile({
    checkpointer,
    interruptBefore: ['human_review'],  // 在该节点执行前暂停
    // 或 interruptAfter: ['generate_draft']  // 在该节点执行后暂停

    # 在 Express API 里实现审核接口

    // 第一步：启动工作流
app.post('/api/workflow/start', async (req, res) => {
  const { topic } = req.body
  const threadId = `thread_${Date.now()}`
  const config = { configurable: { thread_id: threadId } }

  // 运行到中断点停止
  await workflow.invoke({ topic }, config)

  const state = await workflow.getState(config)
  res.json({
    threadId,
    status: 'waiting_review',
    draft: state.values.draft,
  })
})

// 第二步：提交审核结果
app.post('/api/workflow/review', async (req, res) => {
  const { threadId, feedback, action } = req.body  // action: 'approve' | 'reject'
  const config = { configurable: { thread_id: threadId } }

  if (action === 'reject') {
    // 拒绝：注入修改意见，重新执行
    await workflow.updateState(config, { humanFeedback: feedback })
    const result = await workflow.invoke(null, config)
    res.json({ status: 'revised', content: result.revision })
  } else {
    // 通过：直接跳到发布
    await workflow.updateState(config, { humanFeedback: 'ok' })
    const result = await workflow.invoke(null, config)
    res.json({ status: 'published', publishedAt: result.publishedAt })
  }
})

    # 8.7 流式推送节点状态到前端

    工作流执行过程中，用 SSE 实时推送每个节点的状态变化：

    // 服务端：在节点函数内部主动推送事件
function buildWorkflow(sendEvent) {
  async function outlineNode(state) {
    sendEvent('node_start', { node: 'outline', label: '大纲生成' })

    const result = await generateOutline(state.topic)

    sendEvent('node_done', { node: 'outline', result: result.slice(0, 80) + '...' })
    return { outline: result }
  }

  async function draftNode(state) {
    sendEvent('node_start', { node: 'draft', label: '内容撰写' })

    // 流式生成：逐 token 推送
    let draft = ''
    const stream = await draftChain.stream({ topic: state.topic, outline: state.outline })
    for await (const chunk of stream) {
      draft += chunk
      sendEvent('token', { node: 'draft', token: chunk })
    }

    sendEvent('node_done', { node: 'draft', result: `${draft.length}字` })
    return { draft }
  }

  // ... 构建并返回 workflow
}

app.post('/api/workflow/run', async (req, res) => {
  res.setHeader('Content-Type', 'text/event-stream')
  res.setHeader('Cache-Control', 'no-cache')

  const send = (event, data) =>
    res.write(`event: ${event}\ndata: ${JSON.stringify(data)}\n\n`)

  const workflow = buildWorkflow(send)
  send('start', { topic: req.body.topic })

  await workflow.invoke({ topic: req.body.topic }, config)

  send('complete', { ... })
  res.end()
})

async function runWorkflow(topic) {
  const res = await fetch('/api/workflow/run', {
    method: 'POST', headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ topic }),
  })

  const reader = res.body.getReader()
  const decoder = new TextDecoder()
  let buffer = ''

  while (true) {
    const { value, done } = await reader.read()
    if (done) break

    buffer += decoder.decode(value, { stream: true })
    const parts = buffer.split('\n\n')
    buffer = parts.pop() ?? ''

    for (const part of parts) {
      const lines = part.split('\n')
      let event = '', dataStr = ''
      for (const line of lines) {
        if (line.startsWith('event: ')) event = line.slice(7)
        if (line.startsWith('data: '))  dataStr = line.slice(6)
      }
      if (!dataStr) continue
      const data = JSON.parse(dataStr)

      if (event === 'node_start') {
        const node = flowNodes.find(n => n.id === data.node)
        if (node) node.status = 'running'
      }

      if (event === 'node_done') {
        const node = flowNodes.find(n => n.id === data.node)
        if (node) { node.status = 'done'; node.result = data.result }
      }

      if (event === 'token') {
        draftStream.value += data.token  // 实时展示 draft 节点的流式内容
      }
    }
  }
}

    # 8.9 本章小结

    • Workflow 适合步骤已知、可重复、需要可控的任务；Agent 适合路径不确定、需要自主规划的任务
    • 线性工作流：固定顺序，每步读取上一步写入的状态字段
    • 条件工作流：addConditionalEdges 根据状态动态选路，配合扇出+扇入实现多路并汇
    • 并行工作流：从同一前置节点出发到多个节点实现并行，汇聚节点等所有前置完成后执行；RunnableParallel 是更轻量的无状态并行方案
    • Human-in-the-Loop：interruptBefore 暂停工作流，updateState 注入人工输入，invoke(null, config) 继续执行
    • 状态持久化：MemorySaver 用于开发，生产换 PostgresSaver 或 RedisSaver；持久化后支持断点恢复和历史回溯
    • 前端可视化：在节点函数内主动调用 sendEvent 推送状态，前端接收 node_start / node_done / token 事件更新 UI
