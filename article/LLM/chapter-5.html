    # 五、MCP 协议（Model Context Protocol）

    MCP 是 Anthropic 在 2024 年底发布的开放协议，目标是把 AI 工具调用标准化。第四章的 Function Call 是"你自己写工具、自己用"，MCP 是"按标准写工具，所有 AI 应用都能用"。

    # 5.1 MCP 解决什么问题

    没有 MCP 之前，每个 AI 应用都要自己写工具调用逻辑：

    ChatGPT Plugin   → 一套接口规范
Claude Tool Use  → 另一套接口规范
你自己的应用     → 又一套自定义实现
    1
    2
    3

    同一个"查 GitHub Issues"的功能，接入三个不同的 AI 应用要写三次。

    有了 MCP 之后：

    一个 MCP Server（按标准协议开发）
    ↓
Claude Desktop、Cursor、你自己的应用 → 都能直接接入
    1
    2
    3

    类比：MCP 是 AI 工具调用领域的 USB-C——接口统一了，设备和充电器可以自由组合。

    # MCP 的三种原语

    原语 作用 类比
    Tools（工具） 执行操作，有副作用 函数调用
    Resources（资源） 暴露数据，只读 文件/数据库记录
    Prompts（提示词） 预定义提示词模板 可复用的 Prompt

    # 两种传输方式

    stdio（标准输入输出）
  适合：本地工具、命令行工具
  通信：进程间通过 stdin/stdout 传 JSON-RPC 消息

HTTP + SSE（Server-Sent Events）
  适合：远程部署、多客户端
  通信：HTTP POST 发请求，SSE 推送响应

    # 5.2 写一个 MCP Server

    # 5.2.1 stdio 传输（本地工具）

    // server.js
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js'
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js'
import { z } from 'zod'

const server = new McpServer({
  name: 'frontend-tools',  // Server 名称，客户端显示用
  version: '1.0.0',
})

// 注册工具
server.tool(
  'check_code_style',                              // 工具名
  '检查前端代码风格，返回发现的问题和修复建议',   // 描述（给 AI 看的）
  {
    // 参数 Schema，用 Zod 定义
    code: z.string().describe('要检查的代码片段'),
    lang: z.enum(['javascript', 'typescript', 'vue', 'react'])
      .default('javascript'),
  },
  async ({ code, lang }) => {
    const issues = []

    if (code.includes('var ')) {
      issues.push({ rule: 'no-var', severity: 'error', message: '用 let/const 替代 var' })
    }
    if (code.includes('console.log')) {
      issues.push({ rule: 'no-console', severity: 'warning', message: '生产代码移除 console.log' })
    }

    // 工具返回值必须是这个格式
    return {
      content: [{
        type: 'text',
        text: JSON.stringify({ issueCount: issues.length, issues }, null, 2),
      }],
    }
  }
)

// 注册资源（只读数据）
server.resource(
  'vue3-cheatsheet',
  'file:///docs/vue3-cheatsheet',
  'Vue3 常用 API 速查表',
  'text/markdown',
  async () => ({
    contents: [{
      uri: 'file:///docs/vue3-cheatsheet',
      mimeType: 'text/markdown',
      text: '# Vue3 API 速查\n\n## 响应式\n- ref()\n- reactive()\n...',
    }],
  })
)

// 注册提示词模板
server.prompt(
  'code-review',
  '代码审查提示词模板',
  {
    code: z.string(),
    focus: z.string().optional(),
  },
  async ({ code, focus }) => ({
    messages: [{
      role: 'user',
      content: {
        type: 'text',
        text: `审查这段代码${focus ? `，重点关注：${focus}` : ''}：\n\n${code}`,
      },
    }],
  })
)

// 启动（stdio 模式）
const transport = new StdioServerTransport()
await server.connect(transport)

    # 5.2.2 HTTP + SSE 传输（远程服务）

    // server-http.js
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js'
import { SSEServerTransport } from '@modelcontextprotocol/sdk/server/sse.js'
import express from 'express'

const app = express()
const server = new McpServer({ name: 'remote-tools', version: '1.0.0' })

// 注册工具（和 stdio 版完全一样）
server.tool('get_npm_info', '查询 npm 包信息', { ... }, async ({ packageName }) => { ... })

// 维护客户端连接
const transports = new Map()  // sessionId → transport

// SSE 连接端点：客户端连接时调用
app.get('/sse', async (req, res) => {
  const transport = new SSEServerTransport('/messages', res)
  transports.set(transport.sessionId, transport)

  res.on('close', () => transports.delete(transport.sessionId))

  await server.connect(transport)
})

// 消息接收端点：客户端发请求时调用
app.post('/messages', async (req, res) => {
  const transport = transports.get(req.query.sessionId)
  if (!transport) return res.status(404).json({ error: 'Session not found' })
  await transport.handlePostMessage(req, res)
})

app.listen(3001)

    # 5.4 把 MCP 工具接入 LangChain.js

    MCP Client 获取到的工具需要转成 LangChain 的 tool() 格式，模型才能通过 Function Call 调用：

    import { tool } from '@langchain/core/tools'
import { z } from 'zod'

// 从 MCP Server 动态获取工具，转成 LangChain 格式
async function loadMcpTools(client) {
  const { tools: mcpTools } = await client.listTools()

  return mcpTools.map(mcpTool => {
    // 把 MCP 的 JSON Schema 转成 Zod Schema
    const schema = jsonSchemaToZod(mcpTool.inputSchema)

    return tool(
      async (args) => {
        const result = await client.callTool({
          name: mcpTool.name,
          arguments: args,
        })
        return result.content.map(c => c.text || '').join('\n')
      },
      {
        name: mcpTool.name,
        description: mcpTool.description,
        schema,
      }
    )
  })
}

// 简化版 JSON Schema → Zod 转换
function jsonSchemaToZod(jsonSchema) {
  if (!jsonSchema?.properties) return z.object({})

  const props = {}
  const required = new Set(jsonSchema.required || [])

  for (const [key, prop] of Object.entries(jsonSchema.properties)) {
    let zodType
    if (prop.type === 'string')  zodType = prop.enum ? z.enum(prop.enum) : z.string()
    if (prop.type === 'number')  zodType = z.number()
    if (prop.type === 'boolean') zodType = z.boolean()
    if (prop.type === 'array')   zodType = z.array(z.string())
    if (!zodType) zodType = z.any()

    if (prop.description) zodType = zodType.describe(prop.description)
    if (!required.has(key)) zodType = zodType.optional()
    props[key] = zodType
  }

  return z.object(props)
}

// 使用
const lcTools = await loadMcpTools(client)
const modelWithTools = model.bindTools(lcTools)

const result = await modelWithTools.invoke([
  new HumanMessage('帮我检查一下这段代码：var x = 1'),
])

    # 5.5 在 Express 应用里管理多个 MCP Server

    实际项目往往需要连接多个 MCP Server，比如一个提供代码工具、一个提供数据库查询、一个提供文件操作：

    class McpManager {
  constructor() {
    this.clients = new Map()
    this.allTools = []
  }

  // 连接 stdio Server
  async connectStdio(name, command, args) {
    const transport = new StdioClientTransport({ command, args })
    const client = new Client({ name: `host-${name}`, version: '1.0.0' })
    await client.connect(transport)
    this.clients.set(name, client)
    return client
  }

  // 连接 SSE Server
  async connectSSE(name, url) {
    const transport = new SSEClientTransport(new URL(url))
    const client = new Client({ name: `host-${name}`, version: '1.0.0' })
    await client.connect(transport)
    this.clients.set(name, client)
    return client
  }

  // 从所有 Server 加载工具，合并成一个列表
  async loadAllTools() {
    this.allTools = []

    for (const [serverName, client] of this.clients) {
      const { tools } = await client.listTools()

      for (const mcpTool of tools) {
        const lcTool = tool(
          async (args) => {
            const result = await client.callTool({ name: mcpTool.name, arguments: args })
            return result.content.map(c => c.text || '').join('\n')
          },
          {
            name: `${serverName}_${mcpTool.name}`,  // 加前缀防止命名冲突
            description: `[${serverName}] ${mcpTool.description}`,
            schema: jsonSchemaToZod(mcpTool.inputSchema),
          }
        )
        this.allTools.push(lcTool)
      }
    }

    return this.allTools
  }

  async disconnect() {
    for (const client of this.clients.values()) {
      await client.close()
    }
  }
}

// 应用启动时初始化
const mcpManager = new McpManager()

async function bootstrap() {
  // 连接本地工具 Server
  await mcpManager.connectStdio('code-tools', 'node', ['./mcp-servers/code-tools.js'])

  // 连接远程工具 Server
  await mcpManager.connectSSE('data-api', 'http://internal-api.company.com/mcp/sse')

  // 加载所有工具，构建 LangGraph
  const tools = await mcpManager.loadAllTools()
  // ... 构建 chatGraph
}

// 优雅退出时断开连接
process.on('SIGINT', async () => {
  await mcpManager.disconnect()
  process.exit(0)
})

    # 5.6 Vue3 展示 MCP 工具列表

    <template>
  <div class="mcp-panel">
    <!-- 按 server 分组展示工具 -->
    <div v-for="group in toolGroups" :key="group.server">
      <div class="group-header">
        <span class="status-dot connected" />
        {{ group.server }}（{{ group.tools.length }} 个工具）
      </div>

      <div
        v-for="t in group.tools"
        :key="t.name"
        class="tool-item"
        @click="selectTool(t)"
      >
        <span class="tool-name">{{ t.displayName }}</span>
        <span class="tool-desc">{{ t.description }}</span>
      </div>
    </div>
  </div>
</template>

<script setup>
import { ref, computed, onMounted } from 'vue'

const tools = ref([])

// 从 Express 应用的 /api/tools 端点加载工具列表
onMounted(async () => {
  const res = await fetch('http://localhost:3000/api/tools')
  const { tools: list } = await res.json()

  // 解析工具名前缀，区分来自哪个 MCP Server
  tools.value = list.map(t => {
    const idx = t.name.indexOf('_')
    return {
      ...t,
      server: idx > 0 ? t.name.slice(0, idx) : 'builtin',
      displayName: idx > 0 ? t.name.slice(idx + 1) : t.name,
    }
  })
})

// 按 server 分组
const toolGroups = computed(() => {
  const groups = {}
  for (const t of tools.value) {
    if (!groups[t.server]) groups[t.server] = { server: t.server, tools: [] }
    groups[t.server].tools.push(t)
  }
  return Object.values(groups)
})
</script>

    # React 版本

    // useMcpTools.js
import { useState, useEffect } from 'react'

export function useMcpTools() {
  const [tools, setTools]   = useState([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    fetch('http://localhost:3000/api/tools')
      .then(r => r.json())
      .then(({ tools }) => {
        setTools(tools.map(t => {
          const idx = t.name.indexOf('_')
          return {
            ...t,
            server: idx > 0 ? t.name.slice(0, idx) : 'builtin',
            displayName: idx > 0 ? t.name.slice(idx + 1) : t.name,
          }
        }))
      })
      .finally(() => setLoading(false))
  }, [])

  const toolsByServer = tools.reduce((acc, t) => {
    if (!acc[t.server]) acc[t.server] = []
    acc[t.server].push(t)
    return acc
  }, {})

  return { tools, toolsByServer, loading }
}

// 组件里使用
function McpToolList() {
  const { toolsByServer, loading } = useMcpTools()

  if (loading) return <div>连接 MCP Server...</div>

  return (
    <div>
      {Object.entries(toolsByServer).map(([server, serverTools]) => (
        <div key={server}>
          <h3>{server}（{serverTools.length} 个工具）</h3>
          {serverTools.map(t => (
            <div key={t.name}>
              <strong>{t.displayName}</strong>
              <p>{t.description}</p>
            </div>
          ))}
        </div>
      ))}
    </div>
  )
}

    # 5.8 MCP vs Function Call 的关系

    两者是互补的，不是竞争：

    Function Call MCP
    定义位置 在你的代码里 在独立的 MCP Server 里
    复用性 只有当前应用能用 任何支持 MCP 的客户端都能用
    部署方式 随应用一起部署 独立部署，可以是本地进程或远程服务
    适合场景 业务逻辑紧密结合的工具 通用工具、可复用的数据源

    实际做法：两者结合使用。通用工具（查 npm 包、执行代码、读文件）做成 MCP Server；业务逻辑（查订单、查用户、提交退款）作为 Function Call 直接写在应用里。

    // 混合使用示例
const mcpTools  = await loadMcpTools(client)   // 来自 MCP Server 的通用工具
const bizTools  = [queryOrderTool, submitRefundTool]  // 业务专属工具

const allTools  = [...mcpTools, ...bizTools]
const toolNode  = new ToolNode(allTools)

    # 5.9 本章小结

    • MCP 是 AI 工具调用的标准化协议，一次开发，多个 AI 应用都能接入
    • 三种原语：Tools（执行操作）、Resources（暴露数据）、Prompts（提示词模板）
    • 两种传输：stdio 适合本地工具，HTTP+SSE 适合远程部署
    • Server 端用 @modelcontextprotocol/sdk 的 McpServer 开发，工具定义方式和第四章基本一致
    • Client 端连接后可以列出工具、调用工具、读取资源，然后把工具转成 LangChain 格式供模型使用
    • 生产应用里用 McpManager 管理多个 MCP Server 的连接和工具合并
    • MCP 和 Function Call 互补：通用工具做 MCP，业务工具做 Function Call
